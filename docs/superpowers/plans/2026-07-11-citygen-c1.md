### `map.json` v1.3 meta contract (additive)

```json
{
  "schema_version": "1.3",
  "name": "Valencia, España",
  "zone_id": "z1_M_39_4700_m0_3760",
  "zone_size": "M",
  "stats": {
    "span_x": 2500.0, "span_z": 2500.0,
    "extent_m": [-1250.0, -1250.0, 1250.0, 1250.0],
    "road_km": 41.2, "n_buildings": 8342,
    "largest_component_pct": 97.4, "max_road_slope_pct": 6.1,
    "flags": []
  }
}
```
`flags` possible values: `"sparse_buildings"` (n_buildings < 200), `"steep"` (max_road_slope_pct > 15).

### Zone canonicalization contract

- Sizes: `S=1500.0`, `M=2500.0`, `L=3500.0` m (side length).
- Center quantization: round `lat`/`lon` each to the nearest multiple of `0.002°`.
- bbox from canonical center: `dlat = (side/2) / 110540`, `dlon = (side/2) / (111320 * cos(radians(lat_c)))` → `(lat_c-dlat, lon_c-dlon, lat_c+dlat, lon_c+dlon)`.
- Projection origin = canonical center (same projection formulas as today; Godot never sees lat/lon — unchanged).
- `zone_id = "z%d_%s_%s_%s" % (PIPELINE_MAJOR, size, enc(lat_c), enc(lon_c))`, `enc(v)`: format `%.4f`, replace `"-"` → `"m"`, then `"."` → `"_"`. `PIPELINE_MAJOR = 1`.
- Verified by hand: `enc(4.4380) == "4_4380"`, `enc(-75.2020) == "m75_2020"` (Jordán's own canonical center, sanity-checks against the spec's worked example `z1_M_4_4380_m75_2020`).
- Two points within ~200 m of each other (well under half the quantization step in both axes) must canonicalize to the same `lat_c, lon_c` and therefore the same `zone_id`.
- Legacy Jordán mode: explicit historical `bbox`/`origin` passed straight through (no derivation), `zone_size: "custom"`. `zone_id` still computed by running canonicalization on the historical origin, with size `"custom"` — it's only a cache key, never used for bbox/origin math in this mode.
- Raw per-zone cache: `cache/<zone_id>/{overpass_raw.json,elevation_raw.json}`; legacy Jordán uses `cache/legacy/` instead of a zone_id-keyed dir (contract requirement, keeps today's cache files reachable by directory rename rather than losing them).

### Validation contract (`citygen/validate.py`) — fail with cause, never degrade silently

| Rule | Threshold | Result |
|---|---|---|
| Minimum roads | `road_km >= 3.0` | else `FAIL "insufficient_road_data"` |
| Connectivity | largest connected component (union-find over nodes fused by key `"%.2f|%.2f"`, weighted by length) `>= 60%` | else `FAIL "fragmented_road_graph"` |
| Building cap | `n_buildings <= 25000` | else `FAIL "zone_too_dense"` |
| File cap | `map.json <= 12 MB` | else `FAIL "zone_too_dense"` |
| Sparse buildings | `n_buildings < 200` | PASS, flag `sparse_buildings` |
| Extreme slope | `max_road_slope_pct > 15` | PASS, flag `steep` |

The node-fusion key mirrors `godot/scripts/traffic_graph.gd` (`FUSE_FMT := "%.2f|%.2f"`, no epsilon) exactly — same string format, same rounding — so Python's connectivity check agrees with what Godot's `TrafficGraph` will actually build at runtime.

```python
"""Identidad de zona: canonicalizacion de centro, zone_id, bbox derivado y
proyeccion lat/lon <-> metros. Unica fuente de verdad para "que zona es esta"
y "que sistema de coordenadas usa" (ver ESPEC_SESION_CIUDADES2, contrato 1).
"""
import math
from dataclasses import dataclass

SIZE_SIDES = {"S": 1500.0, "M": 2500.0, "L": 3500.0}
CENTER_QUANT = 0.002
PIPELINE_MAJOR = 1


def clamp(value: float, lo: float, hi: float) -> float:
    return max(lo, min(hi, value))


def project(lat: float, lon: float, origin: tuple) -> tuple:
    lat0, lon0 = origin
    x = (lon - lon0) * 111320.0 * math.cos(math.radians(lat0))
    z = -(lat - lat0) * 110540.0
    return round(x, 2), round(z, 2)


def deproject(x: float, z: float, origin: tuple) -> tuple:
    """Inversa exacta de project(): mundo (x, z) -> (lat, lon)."""
    lat0, lon0 = origin
    lon = lon0 + x / (111320.0 * math.cos(math.radians(lat0)))
    lat = lat0 - z / 110540.0
    return lat, lon


def canonicalize_center(lat: float, lon: float) -> tuple:
    lat_c = round(round(lat / CENTER_QUANT) * CENTER_QUANT, 4)
    lon_c = round(round(lon / CENTER_QUANT) * CENTER_QUANT, 4)
    return lat_c, lon_c


def derive_bbox(lat_c: float, lon_c: float, side: float) -> tuple:
    dlat = (side / 2.0) / 110540.0
    dlon = (side / 2.0) / (111320.0 * math.cos(math.radians(lat_c)))
    return (lat_c - dlat, lon_c - dlon, lat_c + dlat, lon_c + dlon)


def encode_coord(v: float) -> str:
    return f"{v:.4f}".replace("-", "m").replace(".", "_")


def make_zone_id(size: str, lat_c: float, lon_c: float) -> str:
    return f"z{PIPELINE_MAJOR}_{size}_{encode_coord(lat_c)}_{encode_coord(lon_c)}"


@dataclass
class ZoneSpec:
    name: str
    size: str  # "S" | "M" | "L" | "custom" (legado)
    lat: float = None
    lon: float = None
    bbox: tuple = None     # explicito: modo legado (Jordan)
    origin: tuple = None   # explicito: modo legado (Jordan)


@dataclass
class ResolvedZone:
    zone_id: str
    cache_key: str  # subcarpeta de cache/: zone_id normalmente, "legacy" para Jordan
    bbox: tuple
    origin: tuple
    size: str
    name: str


def resolve(spec: ZoneSpec) -> ResolvedZone:
    if spec.bbox is not None or spec.origin is not None:
        if spec.bbox is None or spec.origin is None:
            raise ValueError("ZoneSpec legado requiere bbox Y origin explicitos")
        lat_c, lon_c = canonicalize_center(*spec.origin)
        zone_id = make_zone_id(spec.size, lat_c, lon_c)
        return ResolvedZone(
            zone_id=zone_id, cache_key="legacy",
            bbox=spec.bbox, origin=spec.origin, size=spec.size, name=spec.name,
        )

    if spec.lat is None or spec.lon is None:
        raise ValueError("ZoneSpec normal requiere lat Y lon")
    if spec.size not in SIZE_SIDES:
        raise ValueError(f"size invalido: {spec.size!r} (debe ser S, M o L)")

    lat_c, lon_c = canonicalize_center(spec.lat, spec.lon)
    bbox = derive_bbox(lat_c, lon_c, SIZE_SIDES[spec.size])
    zone_id = make_zone_id(spec.size, lat_c, lon_c)
    return ResolvedZone(
        zone_id=zone_id, cache_key=zone_id,
        bbox=bbox, origin=(lat_c, lon_c), size=spec.size, name=spec.name,
    )
```

```python
def compute_grid_params(bbox, origin, margin: float, cell_size: float) -> dict:
    lat_s, lon_o, lat_n, lon_e = bbox
    corners = [(lat_s, lon_o), (lat_s, lon_e), (lat_n, lon_o), (lat_n, lon_e)]
    projected = [zones.project(lat, lon, origin) for lat, lon in corners]
    xs = [p[0] for p in projected]
    zs = [p[1] for p in projected]
    origin_x = round(min(xs) - margin, 2)
    origin_z = round(min(zs) - margin, 2)
    width = (max(xs) + margin) - origin_x
    height = (max(zs) + margin) - origin_z
    nx = math.ceil(width / cell_size) + 1
    nz = math.ceil(height / cell_size) + 1
    return {"cell_size": cell_size, "nx": nx, "nz": nz, "origin_x": origin_x, "origin_z": origin_z}


def height_at(x: float, z: float, terrain: dict) -> float:
    cell_size = terrain["cell_size"]
    nx = terrain["nx"]; nz = terrain["nz"]
    origin_x = terrain["origin_x"]; origin_z = terrain["origin_z"]
    heights = terrain["heights"]
    gx = zones.clamp((x - origin_x) / cell_size, 0.0, nx - 1.001)
    gz = zones.clamp((z - origin_z) / cell_size, 0.0, nz - 1.001)
    i = int(math.floor(gx)); j = int(math.floor(gz))
    fx = gx - i; fz = gz - j
    h00 = heights[j * nx + i]; h10 = heights[j * nx + i + 1]
    h01 = heights[(j + 1) * nx + i]; h11 = heights[(j + 1) * nx + i + 1]
    top = h00 + (h10 - h00) * fx
    bottom = h01 + (h11 - h01) * fx
    return top + (bottom - top) * fz

def load_elevation_cache(path: str, bbox, grid_params: dict):
    if not os.path.isfile(path):
        return None
    try:
        with open(path, "r", encoding="utf-8") as f:
            cache = json.load(f)
    except (json.JSONDecodeError, OSError):
        return None
    if list(cache.get("bbox", [])) != list(bbox):
        return None
    if cache.get("grid") != grid_params:
        return None
    return cache.get("elevations")


def save_elevation_cache(path: str, bbox, grid_params: dict, elevations) -> None:

def fetch_elevation_batch(locations, retries: int = ELEVATION_RETRIES, backoff=ELEVATION_BACKOFF):
    locations_param = "|".join(f"{lat:.6f},{lon:.6f}" for lat, lon in locations)
    url = f"{OPENTOPODATA_URL}?locations={locations_param}&interpolation=bilinear"
    headers = {"User-Agent": USER_AGENT}
    last_error = None
    for attempt in range(retries + 1):
        try:
            resp = requests.get(url, headers=headers, timeout=60)
            if resp.status_code == 429 or resp.status_code >= 500:
                raise RuntimeError(f"HTTP {resp.status_code}")
            resp.raise_for_status()
            data = resp.json()
            if data.get("status") != "OK":
                raise RuntimeError(f"status API: {data.get('status')}")
            return [r.get("elevation") for r in data["results"]]
        except Exception as exc:
            last_error = exc
            if attempt < retries:
                wait = backoff[min(attempt, len(backoff) - 1)]
                print(f"  lote elevacion fallo (intento {attempt + 1}/{retries + 1}): {exc}; reintento en {wait}s")
                time.sleep(wait)
    raise RuntimeError(f"Lote de elevacion fallo tras {retries} reintentos: {last_error}")


def fetch_elevations(grid_points, grid_params: dict, bbox, cache_dir: str, refresh_elevation: bool):

    cached = None if refresh_elevation else load_elevation_cache(cache_path, bbox, grid_params)
    if cached is not None:
        print(f"Usando cache de elevacion: {cache_path}")
        return cached, None
    num_batches = math.ceil(len(grid_points) / ELEVATION_BATCH_SIZE)
    all_elevations = []
    for batch_idx in range(num_batches):
        start = batch_idx * ELEVATION_BATCH_SIZE

    raw_elevations, num_batches = fetch_elevations(grid_points, grid_params, bbox, cache_dir, refresh_elevation)
    filled, nulls_filled = fill_nulls(raw_elevations, nx, nz)

    max_grade_before_smooth = max_adjacent_grade(filled, nx, nz, cell_size)
    smoothed, nodes_preserved = smooth_heights(filled, nx, nz, SMOOTH_PASSES, SMOOTH_DELTA_MAX)
    max_grade_after_smooth = max_adjacent_grade(smoothed, nx, nz, cell_size)

    raw_min = min(smoothed)
    raw_max = max(smoothed)
    exaggeration = zones.clamp(TERRAIN_EXAGGERATION, 1.0, 1.3)
    exaggerated = [raw_min + (h - raw_min) * exaggeration for h in smoothed]

    base_elevation = round(raw_min, 1)
    heights_relative = [round(h - raw_min, 2) for h in exaggerated]

    terrain = {
        "cell_size": cell_size, "nx": nx, "nz": nz,
        "origin_x": origin_x, "origin_z": origin_z,
        "base_elevation": base_elevation, "heights": heights_relative,
    }
    terrain_stats = {
        "num_points": len(grid_points), "num_batches": num_batches,
        "nulls_filled": nulls_filled, "elevation_min_abs": raw_min, "elevation_max_abs": raw_max,
        "elevation_min_rel": 0.0, "elevation_max_rel": max(heights_relative),
        "max_grade_before_smooth": max_grade_before_smooth, "max_grade_after_smooth": max_grade_after_smooth,
        "nodes_preserved": nodes_preserved,
        "nodes_preserved_pct": (100.0 * nodes_preserved / len(smoothed)) if smoothed else 0.0,
        "exaggeration": exaggeration,
    }
    return terrain, terrain_stats
```

```python
"""Validaciones estructurales del pipeline (contrato 3, ESPEC_SESION_CIUDADES2).
Fallar con causa exacta, nunca degradar en silencio. Union-find espeja
godot/scripts/traffic_graph.gd (FUSE_FMT "%.2f|%.2f", sin epsilon) para que la
conectividad medida aqui coincida con el grafo que Godot construira en runtime.
"""
import math

MIN_ROAD_KM = 3.0
MIN_LARGEST_COMPONENT_PCT = 60.0
MAX_BUILDINGS = 25000
MAX_FILE_BYTES = 12 * 1024 * 1024
SPARSE_BUILDINGS_THRESHOLD = 200
STEEP_SLOPE_THRESHOLD_PCT = 15.0

FUSE_FMT = "%.2f|%.2f"


class _DSU:
    def __init__(self):
        self.parent = {}
        self.size = {}

    def add(self, key):
        if key not in self.parent:
            self.parent[key] = key
            self.size[key] = 1

    def find(self, key):
        while self.parent[key] != key:
            self.parent[key] = self.parent[self.parent[key]]
            key = self.parent[key]
        return key

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return
        if self.size[ra] < self.size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self.size[ra] += self.size[rb]


def compute_connectivity(roads) -> tuple:
    dsu = _DSU()
    edges = []
    for road in roads:
        points = road["points"]
        for i in range(len(points) - 1):
            x0, z0 = points[i]
            x1, z1 = points[i + 1]
            ka = FUSE_FMT % (x0, z0)
            kb = FUSE_FMT % (x1, z1)
            dsu.add(ka)
            dsu.add(kb)
            length = math.hypot(x1 - x0, z1 - z0)
            edges.append((ka, kb, length))
            dsu.union(ka, kb)

    total_length = sum(e[2] for e in edges)
    if total_length <= 0.0:
        return 0.0, 0.0

    component_length = {}
    for ka, _kb, length in edges:
        root = dsu.find(ka)
        component_length[root] = component_length.get(root, 0.0) + length

    largest = max(component_length.values())
    return total_length / 1000.0, 100.0 * largest / total_length


def check_road_data(road_km: float):
    return None if road_km >= MIN_ROAD_KM else "insufficient_road_data"


def check_connectivity(largest_component_pct: float):
    return None if largest_component_pct >= MIN_LARGEST_COMPONENT_PCT else "fragmented_road_graph"


def check_building_cap(n_buildings: int):
    return None if n_buildings <= MAX_BUILDINGS else "zone_too_dense"


def check_file_size(size_bytes: int):
    return None if size_bytes <= MAX_FILE_BYTES else "zone_too_dense"


def compute_flags(n_buildings: int, max_road_slope_pct: float) -> list:
    flags = []
    if n_buildings < SPARSE_BUILDINGS_THRESHOLD:
        flags.append("sparse_buildings")
    if max_road_slope_pct > STEEP_SLOPE_THRESHOLD_PCT:
        flags.append("steep")
    return flags
```

```python
def _append_hillshade(parts, terrain: dict):
    heights = terrain["heights"]
    nx, nz = terrain["nx"], terrain["nz"]
    cell = terrain["cell_size"]
    origin_x, origin_z = terrain["origin_x"], terrain["origin_z"]
    max_h = max(heights) if heights else 0.0
    for j in range(nz - 1):
        for i in range(nx - 1):
            h_avg = (
                heights[j * nx + i] + heights[j * nx + i + 1]
                + heights[(j + 1) * nx + i] + heights[(j + 1) * nx + i + 1]
            ) / 4.0
            gray = int(round(255 * (h_avg / max_h))) if max_h > 0 else 180
            gray = max(0, min(255, gray))
            x = origin_x + i * cell
            z = origin_z + j * cell
            parts.append(
                f'<rect x="{x:.2f}" y="{z:.2f}" width="{cell:.2f}" height="{cell:.2f}" '
                f'fill="rgb({gray},{gray},{gray})" opacity="0.5" />'
            )


def generate_svg(roads, buildings, terrain, path):
    all_x = []
    all_z = []
    for road in roads:
        for x, z in road["points"]:
            all_x.append(x)
            all_z.append(z)
    for b in buildings:
        for x, z in b["footprint"]:
            all_x.append(x)
            all_z.append(z)
    all_x.append(terrain["origin_x"])
    all_x.append(terrain["origin_x"] + (terrain["nx"] - 1) * terrain["cell_size"])
    all_z.append(terrain["origin_z"])
    all_z.append(terrain["origin_z"] + (terrain["nz"] - 1) * terrain["cell_size"])

    if not all_x:
        print("Sin datos para preview.svg")
        return

    margin = 10.0
    min_x, max_x = min(all_x) - margin, max(all_x) + margin
    min_z, max_z = min(all_z) - margin, max(all_z) + margin
    width = max_x - min_x
    height = max_z - min_z

    parts = [
        f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="{min_x:.2f} {min_z:.2f} {width:.2f} {height:.2f}">',
        f'<rect x="{min_x:.2f}" y="{min_z:.2f}" width="{width:.2f}" height="{height:.2f}" fill="#f0f0f0" />',
    ]

    _append_hillshade(parts, terrain)

    for b in buildings:
        pts = " ".join(f"{x:.2f},{z:.2f}" for x, z in b["footprint"])
        parts.append(f'<polygon points="{pts}" fill="#999999" stroke="none" />')

    for road in roads:
        pts = " ".join(f"{x:.2f},{z:.2f}" for x, z in road["points"])
        stroke_width = max(road["width"] * 0.5, 0.5)
        parts.append(
            f'<polyline points="{pts}" fill="none" stroke="#000000" '
            f'stroke-width="{stroke_width:.2f}" stroke-linecap="round" stroke-linejoin="round" />'
        )

    parts.append("</svg>")

    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(parts))
    print(f"preview.svg escrito: {path}")
```

```python
            "provider": fetch_elevation.ELEVATION_PROVIDER,
            "fetched_at": generated_at,
        },
        "name": resolved.name,
        "zone_id": resolved.zone_id,
        "zone_size": resolved.size,
        "stats": stats,
    }


def _extent_and_span(bbox, origin):
    lat_s, lon_o, lat_n, lon_e = bbox
    corners = [(lat_s, lon_o), (lat_s, lon_e), (lat_n, lon_o), (lat_n, lon_e)]
    projected = [zones.project(lat, lon, origin) for lat, lon in corners]
    xs = [p[0] for p in projected]
    zs = [p[1] for p in projected]
    extent = [min(xs), min(zs), max(xs), max(zs)]
    span_x = extent[2] - extent[0]
    span_z = extent[3] - extent[1]
    return extent, span_x, span_z


def generate(zone: "zones.ZoneSpec", cache_dir: str, out_path: str, progress_cb=None,
             refresh: bool = False, refresh_elevation: bool = False, svg_path: str = None) -> GenResult:
    def report(stage):
        if progress_cb is not None:
            progress_cb(stage)

    resolved = zones.resolve(zone)
    zone_cache_dir = os.path.join(cache_dir, resolved.cache_key)

    report("osm")
    raw = fetch_osm.fetch(resolved.bbox, zone_cache_dir, refresh)
    nodes, ways = fetch_osm.index_elements(raw)

    osm_stats = {
        "roads_by_type": {}, "roads_oneway_by_value": {0: 0, 1: 0, -1: 0}, "controls_by_type": {},
        "roads_discarded_few_points": 0, "roads_discarded_outside_bbox": 0,
        "buildings_discarded_not_closed": 0, "buildings_discarded_few_points": 0,
        "buildings_discarded_few_vertices": 0, "buildings_discarded_outside_bbox": 0,
    }
    roads = fetch_osm.build_roads(ways, nodes, resolved.bbox, resolved.origin, osm_stats)
    buildings = fetch_osm.build_buildings(ways, nodes, resolved.bbox, resolved.origin, osm_stats)
    controls = fetch_osm.build_controls(raw, resolved.bbox, resolved.origin, osm_stats)

    road_km, largest_component_pct = validate.compute_connectivity(roads)

    reason = validate.check_road_data(road_km) or validate.check_connectivity(largest_component_pct)
    if reason is not None:
        return GenResult(ok=False, reason=reason, stats=None, zone_id=resolved.zone_id)

    reason = validate.check_building_cap(len(buildings))
    if reason is not None:
        return GenResult(ok=False, reason=reason, stats=None, zone_id=resolved.zone_id)

    report("elevation")
    terrain, terrain_stats = fetch_elevation.build_terrain(
        resolved.bbox, resolved.origin, zone_cache_dir, refresh_elevation,
    )

    report("build")
    max_road_slope_pct = 0.0
    for road in roads:
        slope = fetch_elevation.road_slope_stats(road["points"], terrain)
        if slope > max_road_slope_pct:
            max_road_slope_pct = slope

    extent, span_x, span_z = _extent_and_span(resolved.bbox, resolved.origin)
    flags = validate.compute_flags(len(buildings), max_road_slope_pct)
    stats = {
        "span_x": round(span_x, 1), "span_z": round(span_z, 1),
        "extent_m": [round(v, 1) for v in extent],
        "road_km": round(road_km, 1), "n_buildings": len(buildings),
        "largest_component_pct": round(largest_component_pct, 1),
        "max_road_slope_pct": round(max_road_slope_pct, 1), "flags": flags,
    }

    meta = build_meta(resolved, stats)
    map_data = {"meta": meta, "roads": roads, "buildings": buildings, "controls": controls, "terrain": terrain}
    serialized = json.dumps(map_data, ensure_ascii=False, separators=(",", ":"))
    size_bytes = len(serialized.encode("utf-8"))

    reason = validate.check_file_size(size_bytes)
    if reason is not None:
        return GenResult(ok=False, reason=reason, stats=stats, zone_id=resolved.zone_id)

    report("package")
    os.makedirs(os.path.dirname(out_path) or ".", exist_ok=True)
    with open(out_path, "w", encoding="utf-8") as f:
        f.write(serialized)

    if svg_path is not None:
        svg_preview.generate_svg(roads, buildings, terrain, svg_path)

    diagnostics = {
        "osm_stats": osm_stats, "terrain_stats": terrain_stats,
        "size_bytes": size_bytes, "n_roads": len(roads), "n_controls": len(controls),
    }
    return GenResult(ok=True, reason=None, stats=stats, zone_id=resolved.zone_id, diagnostics=diagnostics)
```

```python
#!/usr/bin/env python3
"""CLI de desarrollo sobre citygen: genera map.json para Jordan (legado) o
para cualquier zona nueva (centro + tamano).

Uso:
  python generate_map.py --legacy-jordan [--refresh] [--refresh-elevation] [--stats] [--svg]
  python generate_map.py --lat 4.60 --lon -74.08 --size M --name "Bogota" [--refresh] [--stats] [--svg]
"""
import argparse
import os
import shutil
import time

from citygen import ZoneSpec, generate

SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__))
CACHE_DIR = os.path.join(SCRIPT_DIR, "cache")
OUTPUT_DIR = os.path.join(SCRIPT_DIR, "output")
OUTPUT_JSON_PATH = os.path.join(OUTPUT_DIR, "map.json")
OUTPUT_SVG_PATH = os.path.join(OUTPUT_DIR, "preview.svg")
GODOT_DATA_PATH = os.path.join(SCRIPT_DIR, "..", "godot", "data", "map.json")

# Bbox/origen historicos de Jordan (ESPEC_SESION.md original); se mantienen
# intactos para no huerfanar los tiempos record ya guardados.
LEGACY_BBOX = (4.433220, -75.210321, 4.442772, -75.193563)
LEGACY_ORIGIN = (4.4380, -75.2019)


def _print_stage(stage: str):
    print(f"[{stage}] ...")


def main():
    parser = argparse.ArgumentParser(description="Genera map.json vía citygen.")
    parser.add_argument("--legacy-jordan", action="store_true", help="Regenera Jordan con su bbox/origen historicos.")
    parser.add_argument("--lat", type=float, help="Latitud del centro de la zona nueva.")
    parser.add_argument("--lon", type=float, help="Longitud del centro de la zona nueva.")
    parser.add_argument("--size", choices=["S", "M", "L"], default="M", help="Tamano de zona (default M).")
    parser.add_argument("--name", default="", help="Nombre de presentacion de la zona.")
    parser.add_argument("--refresh", action="store_true", help="Ignora la cache de Overpass.")
    parser.add_argument("--refresh-elevation", action="store_true", help="Ignora la cache de elevacion.")
    parser.add_argument("--stats", action="store_true", help="Imprime estadisticas del mapa generado.")
    parser.add_argument("--svg", action="store_true", help="Genera output/preview.svg.")
    args = parser.parse_args()

    if args.legacy_jordan:
        zone = ZoneSpec(name="Jordán, Ibagué", size="custom", bbox=LEGACY_BBOX, origin=LEGACY_ORIGIN)
    elif args.lat is not None and args.lon is not None:
        if not args.name:
            parser.error("--name es requerido junto con --lat/--lon")
        zone = ZoneSpec(name=args.name, size=args.size, lat=args.lat, lon=args.lon)
    else:
        parser.error("Debe especificar --legacy-jordan o --lat/--lon (+ --size/--name)")

    os.makedirs(OUTPUT_DIR, exist_ok=True)
    svg_path = OUTPUT_SVG_PATH if args.svg else None

    t0 = time.time()
    result = generate(
        zone, CACHE_DIR, OUTPUT_JSON_PATH, progress_cb=_print_stage,
        refresh=args.refresh, refresh_elevation=args.refresh_elevation, svg_path=svg_path,
    )
    elapsed = time.time() - t0

    if not result.ok:
        print(f"FALLO ({elapsed:.1f}s): {result.reason}")
        raise SystemExit(1)

    print(f"map.json escrito en {elapsed:.1f}s: zone_id={result.zone_id}")
    print(
        "  vias=%d edificios=%d controles=%d road_km=%.1f componente_mayor=%.1f%% "
        "pendiente_max=%.1f%% flags=%s"
        % (
            result.diagnostics["n_roads"], result.stats["n_buildings"], result.diagnostics["n_controls"],
            result.stats["road_km"], result.stats["largest_component_pct"],
            result.stats["max_road_slope_pct"], result.stats["flags"],
        )
    )

    if args.legacy_jordan:
        os.makedirs(os.path.dirname(GODOT_DATA_PATH), exist_ok=True)
        shutil.copyfile(OUTPUT_JSON_PATH, GODOT_DATA_PATH)
        print(f"Copiado a {GODOT_DATA_PATH}")

    if args.stats:
        ts = result.diagnostics["terrain_stats"]
        os_ = result.diagnostics["osm_stats"]
        print("--- stats ---")
        print(f"tamano map.json: {result.diagnostics['size_bytes']} bytes")
        print(f"elevacion: {ts['elevation_min_abs']:.1f}-{ts['elevation_max_abs']:.1f} msnm")
        print(f"pendiente celdas adyacentes: {ts['max_grade_before_smooth']:.1f}% -> {ts['max_grade_after_smooth']:.1f}%")
        print(f"descartes vias: pocos_puntos={os_['roads_discarded_few_points']} fuera_bbox={os_['roads_discarded_outside_bbox']}")
        print(f"descartes edificios: no_cerrado={os_['buildings_discarded_not_closed']} pocos_puntos={os_['buildings_discarded_few_points']} pocos_vertices={os_['buildings_discarded_few_vertices']} fuera_bbox={os_['buildings_discarded_outside_bbox']}")


if __name__ == "__main__":
    main()
```
